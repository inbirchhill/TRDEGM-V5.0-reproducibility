"""모델 확장(Model II, 각도강성 포함) 재시도: §8.1의 3번째 반증 시도(순수 반경모델에서
이웃껍질 강체회전의 복원력이 0에 가까움, 목표비 1/sqrt(2)=0.707과 불일치)를 각도항이
추가된 에너지범함수로 재검정한다. lam6=0.8, lam12=0.0(§8.3 개입실험에서 구리류
탄성비를 가장 잘 맞춘 조합)을 그대로 재사용."""
import numpy as np
from network_base import build_growing_shell, build_multicell_graph, classify_hub_types, build_hub_graph

depth, buf = 2, 10
all_pts, vor, cluster_idx = build_growing_shell(buffer_radius=buf, depth=depth)
G, face_seen, vertex_coord = build_multicell_graph(all_pts, vor, cluster_idx)
hub_pos, hub_N, n_mixed = classify_hub_types(G)
H = build_hub_graph(G, hub_pos, hub_N)
nodes = list(H.nodes())
idx = {n: i for i, n in enumerate(nodes)}
n_hubs = len(nodes)
pos = np.array([hub_pos[n] for n in nodes])
types = np.array([hub_N[n] for n in nodes])
deg = dict(H.degree())
edges = np.array([[idx[a], idx[b]] for a, b in H.edges()])
print(f"전체 노드 {n_hubs}, 엣지 {len(edges)}", flush=True)

q, p_exp, eps = 4.0, 8.0, 1.0
def dFdrho(rho): return -eps*0.5/np.sqrt(rho)
def calibrate_A(ri, rj): return -(2*q/p_exp)*(dFdrho(ri)+dFdrho(rj))
RHO = {6: 10, 12: 20}
def r0_of(t1, t2):
    if t1 == 6 and t2 == 6: return 1.0
    if t1 == 12 and t2 == 12: return 1.4142136
    return 0.8660254
def A_of(t1, t2): return calibrate_A(RHO[t1], RHO[t2])
r0_edges = np.array([r0_of(types[i], types[j]) for i, j in edges])
A_edges = np.array([A_of(types[i], types[j]) for i, j in edges])
def phi(r, r0): return np.exp(-2*q*(r/r0-1))
def V(r, r0, A): return A*np.exp(-p_exp*(r/r0-1))

# ---- 각도 삼중항 구성 (허브별 이웃쌍 전부, 초기기하의 cos0을 그대로 사용) ----
tri_i, tri_j, tri_k, tri_cos0, tri_lam = [], [], [], [], []
LAM6, LAM12 = 0.8, 0.0
for gi in range(n_hubs):
    nbrs = [idx[m] for m in H.neighbors(nodes[gi])]
    z = len(nbrs)
    lam = LAM6 if types[gi] == 6 else LAM12
    if lam == 0: continue
    for a_ in range(z):
        for b_ in range(a_+1, z):
            gj, gk = nbrs[a_], nbrs[b_]
            relj = pos[gj]-pos[gi]; relk = pos[gk]-pos[gi]
            cos0 = np.dot(relj, relk)/(np.linalg.norm(relj)*np.linalg.norm(relk))
            tri_i.append(gi); tri_j.append(gj); tri_k.append(gk)
            tri_cos0.append(cos0); tri_lam.append(lam)
tri_i=np.array(tri_i); tri_j=np.array(tri_j); tri_k=np.array(tri_k)
tri_cos0=np.array(tri_cos0); tri_lam=np.array(tri_lam)
print(f"각도 삼중항 수(lam6={LAM6},lam12={LAM12}): {len(tri_i)}", flush=True)

def energy(pos_arr):
    d = pos_arr[edges[:,0]]-pos_arr[edges[:,1]]
    r = np.linalg.norm(d,axis=1)
    ph = phi(r,r0_edges)
    rho = np.zeros(n_hubs)
    np.add.at(rho, edges[:,0], ph); np.add.at(rho, edges[:,1], ph)
    rho = np.maximum(rho,1e-8)
    E = (-eps*np.sqrt(rho)).sum() + V(r,r0_edges,A_edges).sum()
    if len(tri_i)>0:
        a = pos_arr[tri_j]-pos_arr[tri_i]; b = pos_arr[tri_k]-pos_arr[tri_i]
        ra=np.linalg.norm(a,axis=1); rb=np.linalg.norm(b,axis=1)
        c = np.sum(a*b,axis=1)/(ra*rb)
        E += 0.5*np.sum(tri_lam*(c-tri_cos0)**2)
    return E

E0 = energy(pos)
print(f"기준 에너지: {E0:.4f}", flush=True)

def rotate_shell_energy(hub_idx, axis, phi_angle):
    axis = axis/np.linalg.norm(axis)
    a=np.cos(phi_angle/2); b,c,d = -axis*np.sin(phi_angle/2)
    R = np.array([
        [a*a+b*b-c*c-d*d, 2*(b*c+a*d), 2*(b*d-a*c)],
        [2*(b*c-a*d), a*a+c*c-b*b-d*d, 2*(c*d+a*b)],
        [2*(b*d+a*c), 2*(c*d-a*b), a*a+d*d-b*b-c*c]])
    center = pos[hub_idx]
    nbrs = list(H.neighbors(nodes[hub_idx]))
    nbr_idx = [idx[m] for m in nbrs]
    new_pos = pos.copy()
    for ni in nbr_idx:
        new_pos[ni] = center + R @ (pos[ni]-center)
    return energy(new_pos)

deg_arr = np.array([deg[n] for n in nodes])
dist0 = np.linalg.norm(pos, axis=1)
core6 = [i for i in range(n_hubs) if types[i]==6 and deg_arr[i]==10]
core12 = [i for i in range(n_hubs) if types[i]==12 and deg_arr[i]==20]
core6.sort(key=lambda i: dist0[i]); core12.sort(key=lambda i: dist0[i])
hub6i, hub12i = core6[0], core12[0]

axis_31 = np.array([1,1,1.0])
axis_4 = np.array([1.0,0,0])

def fit_K(hub_idx, axis):
    small_degs = np.array([0.2,0.5,1.0,2.0])
    small_rad = np.radians(small_degs)
    Es = np.array([rotate_shell_energy(hub_idx, axis, r) for r in small_rad])
    dE = Es - E0
    coef = np.polyfit(small_rad, dE, 2)
    K = 2*coef[0]
    return K, dE

K6, dE6 = fit_K(hub6i, axis_31)
K12_4, dE12_4 = fit_K(hub12i, axis_4)
K12_3, dE12_3 = fit_K(hub12i, axis_31)
print(f"\n[Model II, lam6={LAM6}, lam12={LAM12}]")
print(f"N6(3겹축) 회전강성 K6 = {K6:.4f}")
print(f"N12(4겹축) 회전강성 K12(4겹) = {K12_4:.4f}")
print(f"N12(3겹축) 회전강성 K12(3겹) = {K12_3:.4f}  (등방성 확인용)")

target = 1/np.sqrt(2)
for K12, name in [(K12_4,"4겹축"),(K12_3,"3겹축")]:
    ratio_uniform_I = np.sqrt(K12/K6) if K6>0 and K12>0 else float('nan')
    ratio_degI = np.sqrt((K12/20)/(K6/10)) if K6>0 and K12>0 else float('nan')
    print(f"  N12({name}): I=1가정 -> 비율={ratio_uniform_I:.4f} | I=배위수가정 -> 비율={ratio_degI:.4f}  (목표={target:.4f})")

# --- 비교: Model I(각도항 없음, lam6=lam12=0) 동일 계산 재실행 ---
print("\n=== 대조: Model I(순수 반경, 각도강성 없음) ===")
tri_lam_zero = np.zeros_like(tri_lam)
def energy_modelI(pos_arr):
    d = pos_arr[edges[:,0]]-pos_arr[edges[:,1]]
    r = np.linalg.norm(d,axis=1)
    ph = phi(r,r0_edges)
    rho = np.zeros(n_hubs)
    np.add.at(rho, edges[:,0], ph); np.add.at(rho, edges[:,1], ph)
    rho = np.maximum(rho,1e-8)
    return (-eps*np.sqrt(rho)).sum() + V(r,r0_edges,A_edges).sum()

E0_I = energy_modelI(pos)
def rotate_shell_energy_I(hub_idx, axis, phi_angle):
    axis = axis/np.linalg.norm(axis)
    a=np.cos(phi_angle/2); b,c,d = -axis*np.sin(phi_angle/2)
    R = np.array([
        [a*a+b*b-c*c-d*d, 2*(b*c+a*d), 2*(b*d-a*c)],
        [2*(b*c-a*d), a*a+c*c-b*b-d*d, 2*(c*d+a*b)],
        [2*(b*d+a*c), 2*(c*d-a*b), a*a+d*d-b*b-c*c]])
    center = pos[hub_idx]
    nbrs = list(H.neighbors(nodes[hub_idx]))
    nbr_idx = [idx[m] for m in nbrs]
    new_pos = pos.copy()
    for ni in nbr_idx:
        new_pos[ni] = center + R @ (pos[ni]-center)
    return energy_modelI(new_pos)

def fit_K_I(hub_idx, axis):
    small_degs = np.array([0.2,0.5,1.0,2.0])
    small_rad = np.radians(small_degs)
    Es = np.array([rotate_shell_energy_I(hub_idx, axis, r) for r in small_rad])
    dE = Es - E0_I
    coef = np.polyfit(small_rad, dE, 2)
    return 2*coef[0]

K6_I = fit_K_I(hub6i, axis_31)
K12_I = fit_K_I(hub12i, axis_4)
print(f"K6(Model I)={K6_I:.6f}  K12(Model I)={K12_I:.6f}")
if K6_I>1e-8 and K12_I>1e-8:
    print(f"Model I 예측비율(I=배위수가정) = {np.sqrt((K12_I/20)/(K6_I/10)):.4f}  (목표 {target:.4f})")
else:
    print("Model I: 회전강성이 사실상 0 (기존 반증과 일치)")
